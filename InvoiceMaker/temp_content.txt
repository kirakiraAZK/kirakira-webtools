<script>
const LS = {
  shop:'im_shop', items:'im_items', emailjs:'im_emailjs', mapping:'im_mapping',
  orders:'im_orders', seq:'im_invoice_seq', sentlog:'im_sentlog'
};
let currentRows = [], currentHeaders = [];

function loadJSON(key, fallback){ try{ return JSON.parse(localStorage.getItem(key)) ?? fallback; }catch(e){ return fallback; } }
function saveJSON(key, val){ localStorage.setItem(key, JSON.stringify(val)); }
const CURRENCY_LOCALE = {
  IDR:{locale:'id-ID', frac:0}, USD:{locale:'en-US', frac:2}, SGD:{locale:'en-SG', frac:2},
  MYR:{locale:'ms-MY', frac:2}, JPY:{locale:'ja-JP', frac:0}, PHP:{locale:'en-PH', frac:2}
};
function getCurrency(){ const s = loadJSON(LS.shop, {}); return s.currency || 'IDR'; }
function formatMoney(n){
  const cur = getCurrency();
  const conf = CURRENCY_LOCALE[cur] || CURRENCY_LOCALE.IDR;
  return new Intl.NumberFormat(conf.locale,{style:'currency',currency:cur,minimumFractionDigits:conf.frac}).format(Math.round(n||0));
}
function toast(msg){ const t=document.getElementById('toast'); t.textContent=msg; t.style.display='block'; setTimeout(()=>t.style.display='none',3000); }
function escapeHtml(s){ return String(s).replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

// ---------- screen navigation ----------
function goTo(screen){
  document.querySelectorAll('[id^="screen-"]').forEach(el=>el.style.display='none');
  document.getElementById('screen-'+screen).style.display='block';
  document.getElementById('screenLabel').textContent = screen.replace('-',' ').toUpperCase();
  if(screen==='item-manager') renderItemManager();
  if(screen==='sheet-config') fillMappingSelects();
}
function updateClock(){
  const now = new Date();
  document.getElementById('liveClock').textContent =
    now.toLocaleDateString('id-ID',{day:'numeric',month:'long',year:'numeric'}) + '  ' + now.toLocaleTimeString('id-ID');
}
setInterval(updateClock, 1000);
updateClock();
function loadCurrencyToForm(){
  document.getElementById('currencySelect').value = getCurrency();
}
function changeCurrency(val){
  const s = loadJSON(LS.shop, {});
  s.currency = val;
  saveJSON(LS.shop, s);
  renderOrders();
  toast('Mata uang diubah ke '+val);
}

// ---------- shop details ----------
function loadShopDetails(){
  const s = loadJSON(LS.shop, {});
  document.getElementById('shopName').value = s.shopName || '';
  document.getElementById('invoicePrefix').value = s.invoicePrefix || 'INV-';
  document.getElementById('paymentInfo').value = s.paymentInfo || '';
}
function handleHeaderImage(evt){
  const file = evt.target.files[0];
  if(!file) return;
  if(file.size > 800*1024){ toast('Ukuran gambar terlalu besar (maks 800KB)'); evt.target.value=''; return; }
  const reader = new FileReader();
  reader.onload = e=>{
    const s = loadJSON(LS.shop, {});
    s.headerImage = e.target.result;
    saveJSON(LS.shop, s);
    toast('Header gambar tersimpan');
  };
  reader.readAsDataURL(file);
}
function saveShopDetails(){
  const s = loadJSON(LS.shop, {});
  s.shopName = document.getElementById('shopName').value.trim();
  s.invoicePrefix = document.getElementById('invoicePrefix').value.trim() || 'INV-';
  s.paymentInfo = document.getElementById('paymentInfo').value.trim();
  saveJSON(LS.shop, s);
  toast('Shop Details tersimpan');
}

// ---------- EmailJS config ----------
function loadEmailjsToForm(){
  const e = loadJSON(LS.emailjs, {});
  document.getElementById('ejsService').value = e.serviceId || '';
  document.getElementById('ejsTemplateInvoice').value = e.templateIdInvoice || '';
  document.getElementById('ejsTemplatePaid').value = e.templateIdPaid || '';
  document.getElementById('ejsPublicKey').value = e.publicKey || '';
  if(e.publicKey){ try{ emailjs.init(e.publicKey); }catch(err){} }
}
function saveEmailjsConfig(){
  const conf = {
    serviceId: document.getElementById('ejsService').value.trim(),
    templateIdInvoice: document.getElementById('ejsTemplateInvoice').value.trim(),
    templateIdPaid: document.getElementById('ejsTemplatePaid').value.trim(),
    publicKey: document.getElementById('ejsPublicKey').value.trim(),
  };
  saveJSON(LS.emailjs, conf);
  if(conf.publicKey){ try{ emailjs.init(conf.publicKey); }catch(err){} }
  document.getElementById('ejsStatus').textContent = 'Tersimpan.';
  toast('EmailJS Config tersimpan');
}
function markStepDone(el){ if(el.value.trim()) el.classList.add('field-ok'); else el.classList.remove('field-ok'); }
function saveGuidedConfig(){

function buildPriceMap(){
  const items = loadJSON(LS.items, []);
  const map = {};
  items.forEach(it=>{
    if(it.variations.length===0){
      map[it.name.trim().toLowerCase()] = it.price;
    } else {
      it.variations.forEach(v=>{
        map[(it.name.trim()+'|'+v.name.trim()).toLowerCase()] = v.price;
      });
    }
  });
  return map;
}

// ---------- spreadsheet upload & mapping ----------
function handleSheetUpload(evt){
  const file = evt.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = e=>{
    const data = new Uint8Array(e.target.result);
    const wb = XLSX.read(data, {type:'array'});
    const sheet = wb.Sheets[wb.SheetNames[0]];
    currentRows = XLSX.utils.sheet_to_json(sheet, {defval:''});
    if(currentRows.length===0){ toast('File kosong/tidak dikenali'); return; }
    currentHeaders = Object.keys(currentRows[0]);
    document.getElementById('sheetSummary').textContent = `${file.name} — ${currentRows.length} baris terbaca.`;
    document.getElementById('sheetConfigBtn').disabled = false;
    goTo('sheet-config');
  };
  reader.readAsArrayBuffer(file);
}
function itemOptionsFromManager(){
  const items = loadJSON(LS.items, []);
  const opts = [];
  items.forEach(it=>{
    if(!it.name) return;
    if(it.variations.length===0){
      opts.push({key: it.name.trim().toLowerCase(), label: it.name});
    } else {
      it.variations.forEach(v=>{
        if(!v.name) return;
        opts.push({key: (it.name.trim()+'|'+v.name.trim()).toLowerCase(), label: it.name+' - '+v.name});
      });
    }
  });
  return opts;
}

function renderItemColumnMapping(){
  const used = [document.getElementById('mapNama').value, document.getElementById('mapEmail').value,
    document.getElementById('mapMailOrder').value, document.getElementById('mapAddress').value].filter(Boolean);
  const remaining = currentHeaders.filter(h=>!used.includes(h));
  const itemOpts = itemOptionsFromManager();
  const optHtml = '<option value="">— Bukan kolom item —</option>' + itemOpts.map(o=>`<option value="${escapeHtml(o.key)}">${escapeHtml(o.label)}</option>`).join('');
  const body = document.getElementById('itemColumnMappingBody');
  body.innerHTML = '';
  if(remaining.length===0){
    body.innerHTML = '<tr><td colspan="2">Tidak ada kolom tersisa untuk dipetakan.</td></tr>';
    return;
  }
  if(itemOpts.length===0){
    body.innerHTML = '<tr><td colspan="2">Item Manager masih kosong — tambah item dulu supaya bisa dipetakan.</td></tr>';
    return;
  }
  remaining.forEach(h=>{
    const guessOpt = itemOpts.find(o=> h.toLowerCase().includes(o.label.toLowerCase()) || o.label.toLowerCase().includes(h.toLowerCase()));
    const tr = document.createElement('tr');
    tr.innerHTML = `<td>${escapeHtml(h)}</td><td><select data-col="${escapeHtml(h)}" class="itemColSelect">${optHtml}</select></td>`;
    body.appendChild(tr);

// ---------- process orders ----------
function processOrders(){
  const colNama=document.getElementById('mapNama').value, colEmail=document.getElementById('mapEmail').value,
        colMailOrder=document.getElementById('mapMailOrder').value, colAddress=document.getElementById('mapAddress').value;
  const colMap = {}; // header -> itemKey
  document.querySelectorAll('.itemColSelect').forEach(sel=>{
    if(sel.value) colMap[sel.dataset.col] = sel.value;
  });
  saveJSON(LS.mapping, {colNama,colEmail,colMailOrder,colAddress,colMap});

  const items = loadJSON(LS.items, []);
  const priceMap = {}, labelMap = {};
  items.forEach(it=>{
    if(it.variations.length===0){
      priceMap[it.name.trim().toLowerCase()] = it.price; labelMap[it.name.trim().toLowerCase()] = it.name;
    } else {
      it.variations.forEach(v=>{
        const k=(it.name.trim()+'|'+v.name.trim()).toLowerCase();
        priceMap[k]=v.price; labelMap[k]=it.name+' - '+v.name;
      });
    }
  });

  const shop = loadJSON(LS.shop, {});
  let seq = loadJSON(LS.seq, 1);
  const sentlog = loadJSON(LS.sentlog, {});
  let skippedNoItems = 0;

  const orders = [];
  currentRows.forEach(row=>{
    const nama = String(row[colNama]||'').trim();
    const email = String(row[colEmail]||'').trim();
    if(!nama || !email) return;
    const mailOrder = colMailOrder ? String(row[colMailOrder]||'').trim() : '';
    const address = colAddress ? String(row[colAddress]||'').trim() : '';

    const lines = [];
    Object.entries(colMap).forEach(([header, itemKey])=>{
      const raw = row[header];
      if(raw===''||raw===undefined||raw===null) return;
      const num = parseFloat(raw);
      if(isNaN(num) || num===0) return; // tidak dipesan
      const qtyValid = Number.isInteger(num) && num>0;
      const matched = priceMap.hasOwnProperty(itemKey);
      const harga = matched ? priceMap[itemKey] : 0;
      const qty = qtyValid ? num : num; // tampilkan apa adanya walau invalid, subtotal tetap dihitung dari nilai mentah untuk referensi
      lines.push({
        itemLabel: labelMap[itemKey] || itemKey,
        qty, qtyValid, harga, matched,
        subtotal: matched ? harga*(qtyValid?qty:0) : 0
      });
    });
    if(lines.length===0){ skippedNoItems++; return; }

    const hasInvalid = lines.some(l=>!l.qtyValid || !l.matched);
    const total = lines.reduce((s,l)=>s+l.subtotal,0);
    const hash = nama.toLowerCase()+'|'+email.toLowerCase()+'|'+total;
    const already = sentlog[hash];
    const invoiceNumber = already ? already.invoiceNumber : (shop.invoicePrefix||'INV-')+new Date().getFullYear()+'-'+String(seq++).padStart(4,'0');
    orders.push({
      nama, email, mailOrder, address, lines, total, hash, invoiceNumber,
      status: already ? 'SENT' : (hasInvalid ? 'NEEDS REVIEW' : 'WAITING'),
      paid:false, paidAt:null, paidEmailStatus:'—'
    });
  });
  saveJSON(LS.seq, seq);
  saveJSON(LS.orders, orders);
  if(skippedNoItems>0) toast(skippedNoItems+' baris dilewati (tidak ada item dipesan)');
  goTo('main');
  renderOrders();
}

function sendInvoice(idx){
  const orders = loadJSON(LS.orders, []);
  const o = orders[idx];
  const e = loadJSON(LS.emailjs, {});
  const shop = loadJSON(LS.shop, {});
  if(!e.serviceId || !e.templateIdInvoice || !e.publicKey){ toast('Lengkapi dulu EmailJS Config'); return; }
  const itemsText = o.lines.map(l=>`- ${l.itemLabel} x${l.qty} = ${formatMoney(l.subtotal)}`).join('\n');
  emailjs.send(e.serviceId, e.templateIdInvoice, {
    to_email:o.email, to_name:o.nama, invoice_number:o.invoiceNumber,
    invoice_date:new Date().toLocaleDateString('id-ID'),
    fulfillment: o.mailOrder && o.mailOrder.toString().toLowerCase()==='true' ? 'Dikirim ke '+o.address : 'Diambil di venue',
    items_text:itemsText, total_formatted:formatMoney(o.total),
    shop_name:shop.shopName||'', payment_info:shop.paymentInfo||''
  }).then(()=>{
    o.status='SENT';
    const sentlog = loadJSON(LS.sentlog, {});
    sentlog[o.hash] = {invoiceNumber:o.invoiceNumber, sentAt:new Date().toISOString()};
    saveJSON(LS.sentlog, sentlog);
    saveJSON(LS.orders, orders);
    renderOrders();
    toast('Terkirim ke '+o.email);
  }).catch(err=>{
    o.status='FAILED';
    saveJSON(LS.orders, orders);
    renderOrders();
    toast('Gagal kirim: '+(err&&err.text?err.text:'cek konsol'));
  });
}
